### 1.3 API Client Pattern
```tsx
// src/shared/services/apiClient.ts
import { QueryClient } from '@tanstack/react-query'
const API_BASE = import.meta.env.VITE_API_BASE_URL
class ApiClient {
  private baseUrl: string
  private token: string | null = null
  constructor(baseUrl: string) {
    this.baseUrl = baseUrl
  }
  setToken(token: string) {
    this.token = token
  }
  private async request(
    method: string,
    path: string,
    body?: unknown,
    params?: Record
  ): Promise<ApiResponse> {
    const url = new URL(`${this.baseUrl}${path}`)
    if (params) {
      Object.entries(params).forEach(([k, v]) => url.searchParams.set(k, v))
    }
    const headers: Record = {
      'Content-Type': 'application/json',
    }
    if (this.token) {
      headers['Authorization'] = `Bearer ${this.token}`
    }
    const response = await fetch(url.toString(), {
      method,
      headers,
      body: body ? JSON.stringify(body) : undefined,
    })
    if (!response.ok) {
      const error = await response.json().catch(() => ({}))
      throw new ApiError(response.status, error.error?.message || 'Request failed', error.error?.code)
    }
    return response.json()
  }
  get(path: string, params?: Record) {
    return this.request('GET', path, undefined, params)
  }
  post(path: string, body?: unknown) {
    return this.request('POST', path, body)
  }
  put(path: string, body?: unknown) {
    return this.request('PUT', path, body)
  }
  delete(path: string) {
    return this.request('DELETE', path)
  }
}
export const api = new ApiClient(API_BASE)
export const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      staleTime: 30_000,
      retry: 3,
      retryDelay: (attempt) => Math.min(1000 * 2 ** attempt, 30000),
      refetchOnWindowFocus: false,
    },
  },
})
```
### 1.4 Zustand Store Pattern
```tsx
// Beispiel: Module Store
import { create } from 'zustand'
import { devtools, persist } from 'zustand/middleware'
interface CampaignState {
  campaigns: Campaign[]
  selectedId: string | null
  filter: CampaignFilter
  // Actions
  setCampaigns: (campaigns: Campaign[]) => void
  selectCampaign: (id: string | null) => void
  setFilter: (filter: Partial) => void
  reset: () => void
}
const initialState = {
  campaigns: [],
  selectedId: null,
  filter: { status: 'all', search: '' },
}
export const useCampaignStore = create()(
  devtools(
    persist(
      (set) => ({
        ...initialState,
        setCampaigns: (campaigns) => set({ campaigns }),
        selectCampaign: (id) => set({ selectedId: id }),
        setFilter: (filter) => set((s) => ({ filter: { ...s.filter, ...filter } })),
        reset: () => set(initialState),
      }),
      { name: 'campaign-store' }
    ),
    { name: 'CampaignStore' }
  )
)
```
