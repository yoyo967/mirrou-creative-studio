## 2. AGENT RUNTIME ENGINE (Herzstück)
### 2.1 Base Agent Pattern
```ts
// src/agents/BaseAgent.ts
import { VertexAI, GenerativeModel } from '@google-cloud/vertexai'
interface AgentConfig {
  id: string
  name: string
  model: string
  temperature: number
  topK: number
  maxTokens: number
  systemPrompt: string
  capabilities: string[]
}
interface AgentTask {
  id: string
  type: string
  input: Record
  context?: Record
  campaign?: string
  priority: 'low' | 'normal' | 'high' | 'critical'
}
interface AgentResult {
  taskId: string
  agentId: string
  output: unknown
  tokens: { input: number; output: number }
  duration: number
  status: 'success' | 'error'
  error?: string
  metadata: Record
}
abstract class BaseAgent {
  protected config: AgentConfig
  protected model: GenerativeModel
  protected db: FirebaseFirestore.Firestore
  constructor(config: AgentConfig, vertexai: VertexAI, db: FirebaseFirestore.Firestore) {
    this.config = config
    this.db = db
    this.model = vertexai.getGenerativeModel({
      model: config.model,
      generationConfig: {
        temperature: config.temperature,
        topK: config.topK,
        maxOutputTokens: config.maxTokens,
      },
      systemInstruction: config.systemPrompt,
    })
  }
  async execute(task: AgentTask): Promise {
    const startTime = Date.now()
    try {
      // 1. Context Assembly
      const context = await this.assembleContext(task)
      // 2. Prompt Construction
      const prompt = this.buildPrompt(task, context)
      // 3. AI Execution
      const response = await this.model.generateContent(prompt)
      const result = response.response
      // 4. Output Processing
      const output = await this.processOutput(result, task)
      // 5. Store Result
      const agentResult: AgentResult = {
        taskId: task.id,
        agentId: this.config.id,
        output,
        tokens: {
          input: result.usageMetadata?.promptTokenCount || 0,
          output: result.usageMetadata?.candidatesTokenCount || 0,
        },
        duration: Date.now() - startTime,
        status: 'success',
        metadata: { model: this.config.model },
      }
      await this.storeResult(agentResult)
      await this.updateStats(agentResult)
      return agentResult
    } catch (error) {
      const agentResult: AgentResult = {
        taskId: task.id,
        agentId: this.config.id,
        output: null,
        tokens: { input: 0, output: 0 },
        duration: Date.now() - startTime,
        status: 'error',
        error: error instanceof Error ? error.message : 'Unknown error',
        metadata: {},
      }
      await this.storeResult(agentResult)
      return agentResult
    }
  }
  protected abstract assembleContext(task: AgentTask): Promise<Record>
  protected abstract buildPrompt(task: AgentTask, context: Record): string
  protected abstract processOutput(result: unknown, task: AgentTask): Promise
  private async storeResult(result: AgentResult): Promise {
    await this.db.collection('agents').doc(this.config.id)
      .collection('history').add({
        ...result,
        timestamp: new Date(),
      })
  }
  private async updateStats(result: AgentResult): Promise {
    const agentRef = this.db.collection('agents').doc(this.config.id)
    await this.db.runTransaction(async (tx) => {
      const doc = await tx.get(agentRef)
      const stats = doc.data()?.stats || { totalTasks: 0, successRate: 1, avgDuration: 0 }
      const total = stats.totalTasks + 1
      const successRate = result.status === 'success'
        ? (stats.successRate * stats.totalTasks + 1) / total
        : (stats.successRate * stats.totalTasks) / total
      const avgDuration = (stats.avgDuration * stats.totalTasks + result.duration) / total
      tx.update(agentRef, {
        'stats.totalTasks': total,
        'stats.successRate': successRate,
        'stats.avgDuration': avgDuration,
        'lastActive': new Date(),
        'status': 'idle',
      })
    })
  }
}
```
### 2.2 Concrete Agent: Content Agent (CC-06)
```ts
// src/agents/ContentAgent.ts
class ContentAgent extends BaseAgent {
  constructor(vertexai: VertexAI, db: FirebaseFirestore.Firestore) {
    super({
      id: 'CC-06',
      name: 'Content Creator',
      model: 'gemini-2.0-pro',
      temperature: 0.8,
      topK: 40,
      maxTokens: 8192,
      systemPrompt: `Du bist CC-06, der Content Creator Agent von AGENTICUM G5.
        Deine Aufgabe ist es, hochwertige Marketing-Inhalte zu erstellen.
        Du schreibst: Blog-Artikel, Social Media Posts, Ad Copy, Email-Kampagnen, Landing Pages.
        Du beachtest IMMER: Brand Voice, Zielgruppe, SEO-Keywords, CTA.
        Du lieferst IMMER strukturierten Output im angeforderten Format.`,
      capabilities: ['blog', 'social', 'email', 'ad-copy', 'landing-page', 'translate'],
    }, vertexai, db)
  }
  protected async assembleContext(task: AgentTask): Promise<Record> {
    const context: Record = {}
    // Brand Voice laden
    if (task.context?.brandId) {
      const brand = await this.db.collection('brands').doc(task.context.brandId as string).get()
      context.brandVoice = brand.data()
    }
    // Campaign Context laden
    if (task.campaign) {
      const campaign = await this.db.collection('campaigns').doc(task.campaign).get()
      context.campaign = campaign.data()
    }
    // Relevante Knowledge Base Chunks laden
    if (task.input.topic) {
      const knowledgeService = new KnowledgeService(this.db)
      context.knowledge = await knowledgeService.semanticSearch(task.input.topic as string, 5)
    }
    return context
  }
  protected buildPrompt(task: AgentTask, context: Record): string {
    const parts: string[] = []
    if (context.brandVoice) {
      parts.push(`BRAND VOICE:\\n${JSON.stringify(context.brandVoice, null, 2)}`)
    }
    if (context.campaign) {
      parts.push(`CAMPAIGN CONTEXT:\\n${JSON.stringify(context.campaign, null, 2)}`)
    }
    if (context.knowledge) {
      parts.push(`RELEVANT KNOWLEDGE:\\n${JSON.stringify(context.knowledge, null, 2)}`)
    }
    parts.push(`TASK: ${task.type}`)
    parts.push(`INPUT:\\n${JSON.stringify(task.input, null, 2)}`)
    parts.push(`OUTPUT FORMAT: Strukturierter JSON mit den Feldern: title, body, excerpt, seo, tags`)
    return parts.join('\\n\\n---\\n\\n')
  }
  protected async processOutput(result: any, task: AgentTask): Promise {
    const text = result.candidates?.[0]?.content?.parts?.[0]?.text || ''
    try {
      // Versuche JSON zu parsen
      const jsonMatch = text.match(/```json\\n?([\\s\\S]*?)\\n?```/) || [null, text]
      return JSON.parse(jsonMatch[1] || text)
    } catch {
      // Fallback: Roher Text
      return { title: '', body: text, excerpt: '', seo: {}, tags: [] }
    }
  }
}
```

### 2.3 Agent Registry & Router
```ts
// src/agents/AgentRegistry.ts
class AgentRegistry {
  private agents: Map = new Map()
  register(agent: BaseAgent): void {
    this.agents.set(agent.config.id, agent)
  }
  get(agentId: string): BaseAgent | undefined {
    return this.agents.get(agentId)
  }
  // Auto-Route: Welcher Agent für welchen Task-Typ?
  route(taskType: string): BaseAgent {
    const routing: Record = {
      'strategy': 'SP-01',
      'planning': 'SP-01',
      'competitive-analysis': 'SP-01',
      'blog': 'CC-06',
      'social-post': 'CC-06',
      'email': 'CC-06',
      'ad-copy': 'CC-06',
      'translate': 'CC-06',
      'data-analysis': 'DA-03',
      'performance-report': 'DA-03',
      'seo-audit': 'DA-03',
      'compliance-check': 'RA-01',
      'risk-assessment': 'RA-01',
      'image-generation': 'VS-02',
      'visual-design': 'VS-02',
      'workflow-execute': 'OP-04',
      'deployment': 'OP-04',
    }
    const agentId = routing[taskType]
    if (!agentId) throw new Error(`No agent registered for task type: ${taskType}`)
    const agent = this.agents.get(agentId)
    if (!agent) throw new Error(`Agent ${agentId} not found in registry`)
    return agent
  }
  listAll(): AgentConfig[] {
    return Array.from(this.agents.values()).map(a => a.config)
  }
}
```

### 2.4 Workflow Execution Engine
```ts
// src/services/workflowService.ts
class WorkflowExecutionEngine {
  private registry: AgentRegistry
  private db: FirebaseFirestore.Firestore
  private pubsub: PubSub
  async executeWorkflow(workflowId: string): Promise {
    const workflow = await this.loadWorkflow(workflowId)
    const run = await this.createRun(workflowId)
    try {
      // Topologisch sortieren (DAG)
      const executionOrder = this.topologicalSort(workflow.nodes, workflow.edges)
      const nodeResults: Map = new Map()
      for (const nodeId of executionOrder) {
        const node = workflow.nodes.find(n => n.id === nodeId)!
        // Input von vorherigen Nodes sammeln
        const inputs = this.collectInputs(node, workflow.edges, nodeResults)
        // Node ausführen
        const result = await this.executeNode(node, inputs, run.id)
        nodeResults.set(nodeId, result)
        // Run aktualisieren
        await this.updateRunNode(run.id, nodeId, result)
        // Event emittieren
        await this.pubsub.topic('workflow-events').publishMessage({
          data: Buffer.from(JSON.stringify({
            type: 'node.completed',
            workflowId, runId: run.id, nodeId,
            result: result.status,
          })),
        })
      }
      await this.completeRun(run.id, 'completed')
      return run
    } catch (error) {
      await this.completeRun(run.id, 'failed', error)
      throw error
    }
  }
  private async executeNode(node: WorkflowNode, inputs: unknown, runId: string): Promise {
    switch (node.type) {
      case 'agent-execute':
        const agent = this.registry.route(node.config.taskType as string)
        return agent.execute({
          id: `${runId}-${node.id}`,
          type: node.config.taskType as string,
          input: { ...node.config, ...inputs },
          priority: 'normal',
        })
      case 'condition-if':
        const condition = this.evaluateCondition(node.config.condition, inputs)
        return { status: 'success', output: { branch: condition ? 'true' : 'false' } }
      case 'action-store':
        await this.storeData(node.config.collection as string, inputs)
        return { status: 'success', output: { stored: true } }
      case 'action-notify':
        await this.sendNotification(node.config, inputs)
        return { status: 'success', output: { notified: true } }
      case 'action-senate':
        return await this.submitToSenate(inputs, node.config)
      case 'delay-wait':
        await this.delay(node.config.duration as number)
        return { status: 'success', output: { waited: true } }
      default:
        throw new Error(`Unknown node type: ${node.type}`)
    }
  }
}
```

```ts
// src/middleware/auth.ts
import { auth } from '../config/firebase'
export const authMiddleware = async (c: Context, next: Next) => {
  const authHeader = c.req.header('Authorization')
  if (!authHeader?.startsWith('Bearer ')) {
    return c.json({ success: false, error: { code: 'UNAUTHORIZED', message: 'Missing auth token' } }, 401)
  }
  try {
    const token = authHeader.split('Bearer ')[1]
    const decoded = await auth.verifyIdToken(token)
    c.set('user', decoded)
    await next()
  } catch (error) {
    return c.json({ success: false, error: { code: 'UNAUTHORIZED', message: 'Invalid auth token' } }, 401)
  }
}
```

```ts
// src/middleware/validation.ts
import { z } from 'zod'
export const validate = (schema: z.ZodSchema) => {
  return async (c: Context, next: Next) => {
    try {
      const body = await c.req.json()
      const validated = schema.parse(body)
      c.set('validatedBody', validated)
      await next()
    } catch (error) {
      if (error instanceof z.ZodError) {
        return c.json({
          success: false,
          error: { code: 'VALIDATION_ERROR', message: 'Invalid request body', details: error.errors }
        }, 400)
      }
      throw error
    }
  }
}
```

### 3.3 Rate Limiter
```ts
// src/middleware/rateLimit.ts
const tokenBuckets = new Map()
export const rateLimiter = (maxRequests: number = 100, windowMs: number = 60000) => {
  return async (c: Context, next: Next) => {
    const userId = c.get('user')?.uid || c.req.header('x-forwarded-for') || 'anonymous'
    const now = Date.now()
    let bucket = tokenBuckets.get(userId)
    if (!bucket || now - bucket.lastRefill > windowMs) {
      bucket = { tokens: maxRequests, lastRefill: now }
    }
    if (bucket.tokens <= 0) {
      return c.json({
        success: false,
        error: { code: 'RATE_LIMITED', message: `Rate limit exceeded. Try again in ${Math.ceil((bucket.lastRefill + windowMs - now) / 1000)}s` }
      }, 429)
    }
    bucket.tokens--
    tokenBuckets.set(userId, bucket)
    c.header('X-RateLimit-Remaining', String(bucket.tokens))
    c.header('X-RateLimit-Reset', String(bucket.lastRefill + windowMs))
    await next()
  }
}
```
